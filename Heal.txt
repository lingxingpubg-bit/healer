local function removeHitbox()
    if hitbox then
        hitbox:Destroy()
        hitbox = nil
    end
end

local function setupHitbox(character)
    removeHitbox()

    if not HitboxEnabled then
        return
    end

    local root = character:WaitForChild("HumanoidRootPart")

    local part = Instance.new("Part")
    part.Name = "VisualHitbox"
    part.Shape = Enum.PartType.Cylinder

    part.Size = Vector3.new(
        HITBOX_HEIGHT,
        HITBOX_DIAMETER,
        HITBOX_DIAMETER
    )

    part.Color = Color3.fromRGB(255, 140, 0)
    part.Transparency = 0.65
    part.Material = Enum.Material.Neon

    part.CanCollide = false
    part.CanTouch = false
    part.CanQuery = false
    part.Massless = true
    part.CastShadow = false

    part.CFrame = root.CFrame *
        CFrame.Angles(0, 0, math.rad(90))

    part.Parent = character

    local weld = Instance.new("WeldConstraint")
    weld.Part0 = root
    weld.Part1 = part
    weld.Parent = part

    hitbox = part
end

local function updateHitbox()
    if HitboxEnabled then
        if currentCharacter then
            setupHitbox(currentCharacter)
        end
    else
        removeHitbox()
    end
end

local function onCharacterAdded(character)
    currentCharacter = character

    if HitboxEnabled then
        setupHitbox(character)
    end

    -- Automatically turn on and latch auto-clicker if player dies
    local humanoid = character:WaitForChild("Humanoid", 5)
    if humanoid then
        humanoid.Died:Connect(function()
            if AutoTriggerSafeZone and not AutoClickEnabled then
                AutoClickEnabled = true
                UpdateAutoClickUI()
            end
        end)
    end
end

end

--==================================================
-- THIRD BOSS SAFE ZONE DISTANCE & CHECK
--==================================================

local function getDistanceToSafeZone()
    local safeZones = workspace:FindFirstChild("thirdBossSafeZones", true)
    if not safeZones then
        return nil, false
    end

    -- Use hitbox position, fallback to HumanoidRootPart
    local checkPos
    if hitbox and hitbox.Parent then
        checkPos = hitbox.Position
    else
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if root then
            checkPos = root.Position
        else
            return nil, false
        end
    end

    local minDistance = math.huge
    local isInside = false

    local zoneParts = safeZones:IsA("BasePart") and {safeZones} or safeZones:GetDescendants()
    for _, part in ipairs(zoneParts) do
        if part:IsA("BasePart") then
            local localPos = part.CFrame:PointToObjectSpace(checkPos)
            local half = part.Size / 2

            -- Bounding box containment check with 2-stud buffer
            if math.abs(localPos.X) <= (half.X + 2)
                and math.abs(localPos.Y) <= (half.Y + 2)
                and math.abs(localPos.Z) <= (half.Z + 2) then
                isInside = true
                minDistance = 0
                break
            end

            -- Closest point on safe zone part surface
            local clamped = Vector3.new(
                math.clamp(localPos.X, -half.X, half.X),
                math.clamp(localPos.Y, -half.Y, half.Y),
                math.clamp(localPos.Z, -half.Z, half.Z)
            )
            local worldClosest = part.CFrame:PointToWorldSpace(clamped)
            local dist = (checkPos - worldClosest).Magnitude

            if dist < minDistance then
                minDistance = dist
            end
        end
    end

    if minDistance == math.huge then
        return nil, false
    end

    return minDistance, isInside
end

--==================================================
-- AUTO CLICK LOOP WITH LATCHING SAFE ZONE & DEATH TRIGGER
--==================================================

task.spawn(function()
    while ScreenGui.Parent do
        local dist, inside = getDistanceToSafeZone()

        -- Check health for instant death trigger
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local isDead = hum ~= nil and hum.Health <= 0

        if isDead and AutoTriggerSafeZone and not AutoClickEnabled then
            AutoClickEnabled = true
            UpdateAutoClickUI()
        end

        -- Update Distance and Status Labels
        if dist == nil then
            ZoneDistLabel.Text = "Safe Zone: Not Found"
            ZoneDistLabel.TextColor3 = Color3.fromRGB(150, 150, 150)
            ZoneStatusLabel.Text = isDead and "Status: DIED (Latched ON)" or "Zone: UNKNOWN"
            ZoneStatusLabel.TextColor3 = isDead and Color3.fromRGB(255, 80, 80) or Color3.fromRGB(150, 150, 150)
        elseif isDead then
            ZoneDistLabel.Text = string.format("Zone Dist: %.1f studs (DEAD)", dist)
            ZoneDistLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
            ZoneStatusLabel.Text = "Status: DIED (Latched ON)"
            ZoneStatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
        elseif inside then
            ZoneDistLabel.Text = "Zone Dist: 0 studs (INSIDE)"
            ZoneDistLabel.TextColor3 = Color3.fromRGB(50, 220, 80)
            ZoneStatusLabel.Text = "Zone: INSIDE (Active)"
            ZoneStatusLabel.TextColor3 = Color3.fromRGB(50, 220, 80)

            -- Automatically turn ON upon entering safe zone
            if AutoTriggerSafeZone and not AutoClickEnabled then
                AutoClickEnabled = true
                UpdateAutoClickUI()
            end
        else
            ZoneDistLabel.Text = string.format("Zone Dist: %.1f studs", dist)
            ZoneDistLabel.TextColor3 = Color3.fromRGB(255, 200, 100)

            if AutoClickEnabled then
                ZoneStatusLabel.Text = "Zone: OUTSIDE (Latched ON)"
                ZoneStatusLabel.TextColor3 = Color3.fromRGB(100, 200, 255)
            else
                ZoneStatusLabel.Text = "Zone: OUTSIDE (Click: OFF)"
                ZoneStatusLabel.TextColor3 = Color3.fromRGB(160, 160, 160)
            end
            -- Note: Remains ON even after stepping outside (latched)
        end

        -- Click Execution
        if AutoClickEnabled then
            local target = LockedPos or UIS:GetMouseLocation()
            VIM:SendMouseButtonEvent(target.X, target.Y, 0, true, game, 0)
            task.wait(0.01)
            VIM:SendMouseButtonEvent(target.X, target.Y, 0, false, game, 0)

            task.wait(AutoClickDelay)
        else
            task.wait(0.05)
        end
    end
end)

--==================================================
-- AUTOMATIC Q/E LOOP
-- Q -> DETECT revitalize -> COOLDOWN
-- E -> DETECT revitalize -> COOLDOWN -> REPEAT
--==================================================

local function waitForRevitalize()
    while AutoEnabled do
        local revitalize = workspace:FindFirstChild("revitalize", true)

        if revitalize then
            print(
                "revitalize detected:",
                revitalize:GetFullName(),
                revitalize.ClassName
            )
            return true
        end

        task.wait(0.1)
    end

    return false
end

local function waitCooldown()
    local elapsed = 0

    while elapsed < Cooldown do
        if not AutoEnabled then
            return false
        end

        local step = math.min(0.1, Cooldown - elapsed)

        task.wait(step)
        elapsed = elapsed + step
    end

    return AutoEnabled
end

task.spawn(function()
    while true do
        if AutoEnabled then

            -- Press Q
            pressKey(Enum.KeyCode.Q)

            -- Wait for revitalize after Q
            if waitForRevitalize() then

                -- Wait for cooldown
                if waitCooldown() then

                    -- Press E
                    pressKey(Enum.KeyCode.E)

                    -- Wait for revitalize after E
                    if waitForRevitalize() then

                        -- Wait for cooldown before repeating
                        waitCooldown()

                    end
                end
            end

        else
            task.wait(0.1)
        end
    end
end)

--==================================================
-- AUTOMATIC TARGET / LOOP TP
--==================================================

task.spawn(function()
    while true do
        task.wait(TeleportInterval)

        if not TeleportEnabled then
            wasTargetAlive = nil
            continue
        end

        local Target = Players:FindFirstChild(TargetName)

        if not Target then
            wasTargetAlive = nil
            continue
        end

        local TargetCharacter = Target.Character

        local TargetHumanoid = TargetCharacter
            and TargetCharacter:FindFirstChildOfClass("Humanoid")

        local TargetRoot = TargetCharacter
            and TargetCharacter:FindFirstChild("HumanoidRootPart")

        local targetAlive = TargetHumanoid ~= nil
            and TargetHumanoid.Health > 0

        -- Reset own character when target dies
        if wasTargetAlive == true and not targetAlive then
            if not resettingCharacter then
                resettingCharacter = true

                local MyCharacter = LocalPlayer.Character

                local MyHumanoid = MyCharacter
                    and MyCharacter:FindFirstChildOfClass("Humanoid")

                if MyHumanoid and MyHumanoid.Health > 0 then
                    MyHumanoid.Health = 0
                end

                task.delay(3, function()
                    resettingCharacter = false
                end)
            end
        end

        wasTargetAlive = targetAlive

        if targetAlive and TargetRoot then
            local MyCharacter = LocalPlayer.Character

            local MyRoot = MyCharacter
                and MyCharacter:FindFirstChild("HumanoidRootPart")

            local MyHumanoid = MyCharacter
                and MyCharacter:FindFirstChildOfClass("Humanoid")

            if MyRoot and MyHumanoid and MyHumanoid.Health > 0 then

                local distance = (
                    MyRoot.Position - TargetRoot.Position
                ).Magnitude

                if distance > FollowDistance then

                    -- Walk toward target
                    MyHumanoid:MoveTo(TargetRoot.Position)

                else

                    -- Teleport directly to target
                    MyRoot.CFrame = TargetRoot.CFrame

                    MyHumanoid:MoveTo(MyRoot.Position)
                end
            end
        end
    end
end)
