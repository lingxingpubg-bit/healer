--==================================================
-- HITBOX
--==================================================

local function removeHitbox()
    if hitbox then
        hitbox:Destroy()
        hitbox = nil
    end
end

local function setupHitbox(character)
    removeHitbox()

    if not HitboxEnabled then
        return
    end

    local root = character:WaitForChild("HumanoidRootPart")

    local part = Instance.new("Part")
    part.Name = "VisualHitbox"
    part.Shape = Enum.PartType.Cylinder

    part.Size = Vector3.new(
        HITBOX_HEIGHT,
        HITBOX_DIAMETER,
        HITBOX_DIAMETER
    )

    part.Color = Color3.fromRGB(255, 140, 0)
    part.Transparency = 0.65
    part.Material = Enum.Material.Neon

    part.CanCollide = false
    part.CanTouch = false
    part.CanQuery = false
    part.Massless = true
    part.CastShadow = false

    part.CFrame = root.CFrame *
        CFrame.Angles(0, 0, math.rad(90))

    part.Parent = character

    local weld = Instance.new("WeldConstraint")
    weld.Part0 = root
    weld.Part1 = part
    weld.Parent = part

    hitbox = part
end

local function updateHitbox()
    if HitboxEnabled then
        if currentCharacter then
            setupHitbox(currentCharacter)
        end
    else
        removeHitbox()
    end
end

-- CHARACTER RESPAWN
--==================================================

local function onCharacterAdded(character)
    currentCharacter = character

    if HitboxEnabled then
        setupHitbox(character)
    end
end

LocalPlayer.CharacterAdded:Connect(onCharacterAdded)

if LocalPlayer.Character then
    onCharacterAdded(LocalPlayer.Character)
end

            startPosition.Y.Offset + delta.Y

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

SliderKnob.Size = UDim2.new(0, 14, 0, 14)

    (Cooldown - MIN_COOLDOWN) / (MAX_COOLDOWN - MIN_COOLDOWN),
    0, 0.5, 0
)
SliderKnob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
SliderKnob.BorderSizePixel = 0
SliderKnob.Parent = SliderBackground

local sliderDragging = false

local function updateSlider(input)
    local relativeX = math.clamp(
        input.Position.X - SliderBackground.AbsolutePosition.X,
        0,
        SliderBackground.AbsoluteSize.X
    )

    local percentage = relativeX / SliderBackground.AbsoluteSize.X

    local rawValue = MIN_COOLDOWN +
        percentage * (MAX_COOLDOWN - MIN_COOLDOWN)

    Cooldown = math.clamp(
        math.floor(rawValue * 2 + 0.5) / 2,
        MIN_COOLDOWN,
        MAX_COOLDOWN
    )

    local normalized = (Cooldown - MIN_COOLDOWN) /
        (MAX_COOLDOWN - MIN_COOLDOWN)

    SliderFill.Size = UDim2.new(normalized, 0, 1, 0)
    SliderKnob.Position = UDim2.new(normalized, 0, 0.5, 0)

    CooldownLabel.Text = "Q/E Cooldown: " ..
        string.format("%.1f", Cooldown) .. "s"
end

SliderBackground.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1

        sliderDragging = true
        updateSlider(input)

end)

SliderKnob.InputBegan:Connect(function(input)

--==================================================
-- AUTOMATIC Q/E LOOP
-- Q -> COOLDOWN -> E -> COOLDOWN -> REPEAT
--==================================================

task.spawn(function()
    while true do
        if AutoEnabled then

            pressKey(Enum.KeyCode.Q)

            local qWait = Cooldown
            local elapsed = 0

            while elapsed < qWait do
                if not AutoEnabled then
                    break
                end

                local step = math.min(0.1, qWait - elapsed)
                task.wait(step)
                elapsed = elapsed + step
            end

            if AutoEnabled then
                pressKey(Enum.KeyCode.E)

                local eWait = Cooldown
                elapsed = 0

                while elapsed < eWait do
                    if not AutoEnabled then
                        break
                    end

                    local step = math.min(0.1, eWait - elapsed)
                    task.wait(step)
                    elapsed = elapsed + step
                end
            end

        else
            task.wait(0.1)
        end
    end
end)

--==================================================
-- AUTOMATIC TARGET / LOOP TP
--==================================================

task.spawn(function()
    while true do
        task.wait(TeleportInterval)

        if not TeleportEnabled then
            wasTargetAlive = nil
            continue
        end

        local Target = Players:FindFirstChild(TargetName)

        if not Target then
            wasTargetAlive = nil
            continue
        end

        local TargetCharacter = Target.Character

        local TargetHumanoid = TargetCharacter
            and TargetCharacter:FindFirstChildOfClass("Humanoid")

        local TargetRoot = TargetCharacter
            and TargetCharacter:FindFirstChild("HumanoidRootPart")

        local targetAlive = TargetHumanoid ~= nil
            and TargetHumanoid.Health > 0

        -- Reset own character when target dies
        if wasTargetAlive == true and not targetAlive then
            if not resettingCharacter then
                resettingCharacter = true

                local MyCharacter = LocalPlayer.Character
                local MyHumanoid = MyCharacter
                    and MyCharacter:FindFirstChildOfClass("Humanoid")

                if MyHumanoid and MyHumanoid.Health > 0 then
                    MyHumanoid.Health = 0
                end

                task.delay(3, function()
                    resettingCharacter = false
                end)
            end
        end

        wasTargetAlive = targetAlive

        if targetAlive and TargetRoot then
            local MyCharacter = LocalPlayer.Character

            local MyRoot = MyCharacter
                and MyCharacter:FindFirstChild("HumanoidRootPart")

            local MyHumanoid = MyCharacter
                and MyCharacter:FindFirstChildOfClass("Humanoid")

            if MyRoot and MyHumanoid and MyHumanoid.Health > 0 then

                local distance = (
                    MyRoot.Position - TargetRoot.Position
                ).Magnitude

                if distance > FollowDistance then

                    -- Walk toward target
                    MyHumanoid:MoveTo(TargetRoot.Position)

                else

                    -- Teleport directly to target
                    MyRoot.CFrame = TargetRoot.CFrame

                    MyHumanoid:MoveTo(MyRoot.Position)
                end
            end
        end
    end
end)
